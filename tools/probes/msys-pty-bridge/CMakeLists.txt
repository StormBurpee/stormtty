cmake_minimum_required(VERSION 3.24)
project(StormTTYBridgeStage2 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Stage2 bridge host requires native Windows")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(STORMTTY_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include("${STORMTTY_SOURCE_ROOT}/cmake/StormTTYGhostty.cmake")
stormtty_import_ghostty()
find_package(Threads REQUIRED)
add_executable(bridge_stage2 stage2-host.cpp "${STORMTTY_SOURCE_ROOT}/src/terminal/terminal.cpp" "${STORMTTY_SOURCE_ROOT}/src/core/shell_context.cpp")
target_include_directories(bridge_stage2 PRIVATE "${STORMTTY_SOURCE_ROOT}/include")
target_link_libraries(bridge_stage2 PRIVATE stormtty::ghostty Threads::Threads)
if(MSVC)
  target_compile_options(bridge_stage2 PRIVATE /W4 /permissive- /utf-8 /EHsc)
elseif(MINGW)
  target_link_options(bridge_stage2 PRIVATE -municode)
endif()
stormtty_copy_ghostty_runtime(bridge_stage2)

# Actual shipped state/console helpers, not substitutes. No app/AI target linkage.
add_executable(stormtty-console-setup "${STORMTTY_SOURCE_ROOT}/src/platform/windows/console_setup.cpp")
add_executable(bridge_managed stage3-host.cpp "${STORMTTY_SOURCE_ROOT}/src/terminal/terminal.cpp"
  "${STORMTTY_SOURCE_ROOT}/src/core/shell_context.cpp" "${STORMTTY_SOURCE_ROOT}/src/platform/windows/shell_state.cpp")
target_include_directories(bridge_managed PRIVATE "${STORMTTY_SOURCE_ROOT}/include")
target_link_libraries(bridge_managed PRIVATE stormtty::ghostty Threads::Threads shell32 ole32 advapi32)
if(MSVC)
  target_compile_options(bridge_managed PRIVATE /W4 /permissive- /utf-8 /EHsc)
elseif(MINGW)
  target_link_options(bridge_managed PRIVATE -municode)
endif()
add_dependencies(bridge_managed stormtty-console-setup)
stormtty_copy_ghostty_runtime(bridge_managed)
