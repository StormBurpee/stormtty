if(APPLE)
    # The host uses AppKit unified toolbars and the native Metal/CoreText stack.
    if(NOT CMAKE_OSX_DEPLOYMENT_TARGET)
        set(CMAKE_OSX_DEPLOYMENT_TARGET "11.0" CACHE STRING "Minimum supported macOS" FORCE)
    elseif(CMAKE_OSX_DEPLOYMENT_TARGET VERSION_LESS "11.0")
        message(FATAL_ERROR "The native macOS host requires macOS 11 or newer")
    endif()
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
    set(CMAKE_OBJCXX_EXTENSIONS OFF)
    # Apple libc++ in Xcode 16.4 lacks the required standard cancellation API.
    # Use a supported matching compiler/header/runtime set (LLVM libc++ 20+),
    # not private experimental feature macros or a replacement cancellation ABI.
    include(CheckCXXSourceCompiles)
    check_cxx_source_compiles("#include <stop_token>
        #include <thread>
        #include <condition_variable>
        #include <mutex>
        int main() {
            std::stop_source source;
            std::condition_variable_any cv;
            std::mutex mutex;
            std::jthread worker([&](std::stop_token token) {
                std::unique_lock lock(mutex);
                cv.wait(lock, token, [] { return false; });
            });
            worker.request_stop();
            source.request_stop();
            return source.get_token().stop_requested() ? 0 : 1;
        }" STORMTTY_HAS_STANDARD_CANCELLATION)
    if(NOT STORMTTY_HAS_STANDARD_CANCELLATION)
        message(FATAL_ERROR "macOS needs a C++20 library with stable stop_token/jthread and cancellable condition_variable_any. Use LLVM/libc++ 20+ with matching headers/runtime for both CXX and OBJCXX; see src/platform/macos/README.md. Do not enable experimental libc++ macros.")
    endif()
endif()

function(stormtty_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

if(BUILD_TESTING)
    function(stormtty_test name source)
        add_executable(${name} ${source})
        target_include_directories(${name} PRIVATE include)
        target_link_libraries(${name} PRIVATE ${ARGN})
        stormtty_warnings(${name})
        stormtty_copy_ghostty_runtime(${name})
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    endfunction()
    if(STORMTTY_TEST_INK)
        find_program(STORMTTY_NODE_EXECUTABLE NAMES node node.exe REQUIRED)
        execute_process(COMMAND "${STORMTTY_NODE_EXECUTABLE}" --version
            RESULT_VARIABLE _node_result OUTPUT_VARIABLE _node_version OUTPUT_STRIP_TRAILING_WHITESPACE)
        string(REGEX REPLACE "^v" "" _node_version "${_node_version}")
        if(NOT _node_result EQUAL 0 OR _node_version VERSION_LESS "22.0.0")
            message(FATAL_ERROR "Ink compatibility tests require Node.js 22 or newer")
        endif()
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/ink/node_modules/ink/package.json")
            message(FATAL_ERROR "Run npm ci --prefix tests/ink --ignore-scripts --no-audit --no-fund before enabling STORMTTY_TEST_INK")
        endif()
        add_executable(ink_session_tests tests/ink_session_tests.cpp)
        target_link_libraries(ink_session_tests PRIVATE stormtty_session)
        stormtty_warnings(ink_session_tests)
        stormtty_copy_ghostty_runtime(ink_session_tests)
        add_test(NAME ink_session_tests COMMAND ink_session_tests
            "${STORMTTY_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/ink/fixture.mjs")
        set_tests_properties(ink_session_tests PROPERTIES TIMEOUT 90 LABELS "ink;native-pty")
    endif()
    stormtty_test(terminal_tests tests/terminal_tests.cpp stormtty_terminal)
    stormtty_test(process_tests tests/process_tests.cpp stormtty_process stormtty_terminal)
    stormtty_test(window_geometry_tests tests/window_geometry_tests.cpp stormtty_workspace)
    stormtty_test(project_workspaces_tests tests/project_workspaces_tests.cpp stormtty_workspace)
    foreach(component workspace commands config)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_${component}_tests.cpp")
            stormtty_test(core_${component}_tests tests/core_${component}_tests.cpp stormtty_workspace)
        endif()
    endforeach()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_tests.cpp")
        stormtty_test(session_tests tests/session_tests.cpp stormtty_session)
    endif()
endif()
