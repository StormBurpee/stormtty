# Can also configure this directory alone: no Ghostty or terminal dependencies.
cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(StormTTYDesktopGlassProbe LANGUAGES CXX)
    option(STORMTTY_EXPERIMENTAL_DESKTOP_GLASS_PROBE "Build desktop capture source probe" OFF)
endif()
if(STORMTTY_EXPERIMENTAL_DESKTOP_GLASS_PROBE)
    if(NOT WIN32 OR NOT MSVC)
        message(FATAL_ERROR "Desktop glass probe requires MSVC and Windows SDK 10.0.22621+")
    endif()
    add_executable(stormtty_desktop_glass_probe main.cpp capture_source.cpp preview_renderer.cpp)
    target_compile_features(stormtty_desktop_glass_probe PRIVATE cxx_std_20)
    target_compile_options(stormtty_desktop_glass_probe PRIVATE /W4 /permissive- /utf-8 /EHsc)
    target_compile_definitions(stormtty_desktop_glass_probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(stormtty_desktop_glass_probe PRIVATE windowsapp d3d11 dxgi d2d1 dxguid dwrite user32 gdi32 ole32)
endif()
