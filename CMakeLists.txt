cmake_minimum_required(VERSION 3.24)
project(StormTTY VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(STORMTTY_BUILD_APP "Build the native application shell" OFF)
include(CTest)
find_package(Threads REQUIRED)
include(cmake/StormTTYGhostty.cmake)
stormtty_import_ghostty()

function(stormtty_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /EHsc)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

add_library(stormtty_workspace src/core/workspace.cpp src/core/workspace_serialization.cpp src/core/commands.cpp src/core/config.cpp)
target_include_directories(stormtty_workspace PUBLIC include)
stormtty_warnings(stormtty_workspace)

add_library(stormtty_terminal src/terminal/terminal.cpp)
target_include_directories(stormtty_terminal PUBLIC include)
target_link_libraries(stormtty_terminal PUBLIC stormtty::ghostty Threads::Threads)
stormtty_warnings(stormtty_terminal)

if(WIN32)
    add_library(stormtty_process src/platform/windows/conpty_process.cpp)
else()
    add_library(stormtty_process src/platform/posix/pty_process.cpp)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        target_link_libraries(stormtty_process PRIVATE util)
    endif()
endif()
target_include_directories(stormtty_process PUBLIC include)
target_link_libraries(stormtty_process PUBLIC Threads::Threads)
stormtty_warnings(stormtty_process)

add_library(stormtty_session src/session/session.cpp)
target_include_directories(stormtty_session PUBLIC include)
target_link_libraries(stormtty_session PUBLIC stormtty_terminal stormtty_process Threads::Threads)
stormtty_warnings(stormtty_session)

if(BUILD_TESTING)
    function(stormtty_test name source)
        add_executable(${name} ${source})
        target_include_directories(${name} PRIVATE include)
        target_link_libraries(${name} PRIVATE ${ARGN})
        stormtty_warnings(${name})
        stormtty_copy_ghostty_runtime(${name})
        add_test(NAME ${name} COMMAND ${name})
        set_tests_properties(${name} PROPERTIES TIMEOUT 90)
    endfunction()
    stormtty_test(terminal_tests tests/terminal_tests.cpp stormtty_terminal)
    stormtty_test(process_tests tests/process_tests.cpp stormtty_process stormtty_terminal)
    stormtty_test(window_geometry_tests tests/window_geometry_tests.cpp stormtty_workspace)
    foreach(component workspace commands config)
        if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/core_${component}_tests.cpp")
            stormtty_test(core_${component}_tests tests/core_${component}_tests.cpp stormtty_workspace)
        endif()
    endforeach()
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/session_tests.cpp")
        stormtty_test(session_tests tests/session_tests.cpp stormtty_session)
    endif()
endif()

if(WIN32)
    add_library(stormtty_renderer src/platform/windows/renderer.cpp src/platform/windows/png_decoder.cpp)
    target_include_directories(stormtty_renderer PUBLIC include)
    target_link_libraries(stormtty_renderer PUBLIC stormtty_terminal PRIVATE d3d11 dxgi d2d1 dwrite dcomp windowscodecs ole32)
    target_compile_definitions(stormtty_renderer PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    stormtty_warnings(stormtty_renderer)
    if(BUILD_TESTING)
        stormtty_test(renderer_smoke tests/renderer_smoke.cpp stormtty_renderer)
        stormtty_test(renderer_png tests/renderer_png.cpp stormtty_renderer)
    endif()
endif()
if(STORMTTY_BUILD_APP)
    if(WIN32)
        add_executable(stormtty WIN32 src/app/windows_main.cpp src/platform/windows/window.cpp)
        target_link_libraries(stormtty PRIVATE stormtty_workspace stormtty_session stormtty_renderer user32 gdi32 dwmapi imm32 ole32 shell32 advapi32 windowscodecs shlwapi)
        target_compile_definitions(stormtty PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
        if(MINGW)
            target_link_options(stormtty PRIVATE -municode)
        endif()
        stormtty_warnings(stormtty)
        stormtty_copy_ghostty_runtime(stormtty)
        if(BUILD_TESTING)
            file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")
            add_test(NAME native_window_smoke COMMAND stormtty --smoke --capture "${CMAKE_BINARY_DIR}/artifacts/native-window.png")
            set_tests_properties(native_window_smoke PROPERTIES TIMEOUT 60)
        endif()
    else()
        message(FATAL_ERROR "The macOS native host is a separate milestone; portable core/PTY tests are available now")
    endif()
endif()
